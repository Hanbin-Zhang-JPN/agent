0. 简介与运行

Python 是一门简洁、易读的解释型高级编程语言,广泛用于 Web 开发、数据科学、人工智能、自动化脚本等领域。它以 缩进 表示代码块,而非花括号。

运行方式

# 1. 直接运行脚本文件
python hello.py

# 2. 进入交互式解释器(REPL)
python

# 3. 单行执行
python -c "print('Hello, World!')"

第一个程序

# hello.py
print("Hello, World!")  # 输出到控制台
提示: Python 使用 # 表示单行注释;多行注释常用三引号 """..."""。缩进推荐使用 4 个空格,同一代码块缩进必须一致。

1. 变量与数据类型

Python 是动态类型语言,变量无需声明类型,直接赋值即可。

# 基本类型
name    = "Alice"        # str  字符串
age     = 25             # int  整数
height  = 1.68           # float 浮点数
is_ok   = True           # bool 布尔 (True / False)
nothing = None           # NoneType 空值
c       = 3 + 4j          # complex 复数

# 多重赋值
x, y, z = 1, 2, 3
a = b = c = 0

# 查看类型 / 类型转换
print(type(age))        # <class 'int'>
num = int("42")          # str -> int
text = str(3.14)         # float -> str
f = float("1.5")         # str -> float
类型示例说明
int10, -3, 0xFF整数,无大小限制
float3.14, 1e-5双精度浮点数
boolTrue, False布尔值(int 子类)
str"hi", 'yo'不可变文本序列
NoneTypeNone表示"无值"

循环

# for 循环遍历可迭代对象
for i in range(5):        # 0,1,2,3,4
    print(i)

for idx, val in enumerate(["a", "b"]):   # 带索引

for a, b in zip([1, 2], ["x", "y"]):  # 并行遍历

# while 循环
n = 0

    n += 1

else:
    print("循环正常结束(未 break)")

match 模式匹配(3.10+)

match command:
    case "start":

6. 函数

# 定义函数,支持类型注解与默认值
def greet(name: str, greeting: str = "你好") -> str:
    """返回问候语(docstring 文档字符串)"""
    return f"{greeting}, {name}!"

greet("小明")                   # 位置参数
greet("小红", greeting="Hi")   # 关键字参数

# *args 收集位置参数, **kwargs 收集关键字参数
def demo(*args, **kwargs):
    print(args)     # 元组
    print(kwargs)   # 字典

demo(1, 2, x=3)          # args=(1,2)  kwargs={'x':3}

# 返回多个值(实为元组)
def min_max(nums):
    return min(nums), max(nums)
lo, hi = min_max([3, 1, 9])

# 匿名函数 lambda
square = lambda x: x ** 2
sorted(data, key=lambda p: p[1])
陷阱: 不要用可变对象作为默认参数(如 def f(x=[])),它只会创建一次并被共享。应使用 None 作默认值,函数内再初始化。

7. 推导式 Comprehension

推导式是 Python 特有的简洁语法,用于快速生成列表、字典、集合。

# 列表推导式
squares = [x ** 2 for x in range(5)]        # [0,1,4,9,16]
evens   = [x for x in range(10) if x % 2 == 0]  # 带过滤条件

# 字典推导式
sq_map = {x: x ** 2 for x in range(4)}   # {0:0, 1:1, 2:4, 3:9}

# 集合推导式
unique = {c for c in "hello"}            # {'h','e','l','o'}

# 生成器表达式(惰性求值,省内存)
gen = (x ** 2 for x in range(1000000))
sum(gen)

8. 面向对象 OOP

class Animal:
    # 类属性(所有实例共享)
    kingdom = "动物界"

    # 构造方法
    def __init__(self, name, sound):
        self.name = name       # 实例属性
        self.sound = sound

    # 实例方法
    def speak(self):
        return f"{self.name} 发出 {self.sound}"

    # 特殊方法:定义 print 时的显示
    def __str__(self):
        return f"Animal({self.name})"

# 继承
class Dog(Animal):
    def __init__(self, name):
        super().__init__(name, "汪汪")   # 调用父类构造

    def speak(self):                 # 方法重写
        return f"{self.name}: 汪!"

d = Dog("旺财")
print(d.speak())        # 旺财: 汪!
print(isinstance(d, Animal))  # True
特殊方法触发时机
__init__创建实例时
__str__str() / print()
__repr__调试显示 / 交互式回显
__len__len(obj)
__eq__== 比较

常用装饰器

class Circle:
    def __init__(self, r):
        self._r = r

    @property            # 把方法变成属性访问
    def area(self):
        return 3.14 * self._r ** 2

    @staticmethod        # 静态方法,不接收 self
    def info():
        return "这是一个圆"

c = Circle(5)
print(c.area)   # 78.5  (无需括号)

9. 异常处理

try:
    result = 10 / 0
except ZeroDivisionError as e:
    print(f"除零错误: {e}")
except (ValueError, TypeError):   # 捕获多种异常
    print("值或类型错误")
else:
    print("没有异常才执行")
finally:
    print("无论如何都执行(常用于清理)")

# 主动抛出异常
def set_age(age):
    if age < 0:
        raise ValueError("年龄不能为负")

# 自定义异常
class MyError(Exception):
    pass
建议: 只捕获你能处理的具体异常类型,避免裸 except: 吞掉所有错误(包括 KeyboardInterrupt)。

11. 文件操作

# 推荐使用 with,自动关闭文件
with open("data.txt", "r", encoding="utf-8") as f:
    content = f.read()          # 读全部

with open("data.txt", encoding="utf-8") as f:
    for line in f:           # 逐行读取(省内存)
        print(line.strip())

# 写入文件
with open("out.txt", "w", encoding="utf-8") as f:
    f.write("第一行\n")
    f.writelines(["a\n", "b\n"])
模式含义
"r"只读(默认)
"w"写入(覆盖原内容)
"a"追加
"x"创建新文件,已存在则报错
"b"二进制模式(如 "rb")

12. 进阶特性

装饰器 Decorator

import time

def timer(func):
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        print(f"耗时 {time.time() - start:.4f}s")
        return result
    return wrapper

@timer                    # 等价于 slow = timer(slow)
def slow():
    time.sleep(1)

生成器 Generator

def countdown(n):
    while n > 0:
        yield n         # yield 暂停并返回值,惰性生成
        n -= 1

for x in countdown(3):   # 3, 2, 1
    print(x)

上下文管理器

from contextlib import contextmanager

@contextmanager
def managed():
    print("进入")
    yield "资源"
    print("退出")

with managed() as r:
    print(r)

类型注解 Type Hints

from typing import Optional

def find(items: list[int], target: int) -> Optional[int]:
    for i, v in enumerate(items):
        if v == target:
            return i
    return None
提示: 类型注解只是提示,运行时不强制检查。配合 mypy 等工具可做静态类型检查。
